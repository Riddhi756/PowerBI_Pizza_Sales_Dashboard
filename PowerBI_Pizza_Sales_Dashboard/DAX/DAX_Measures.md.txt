# DAX Measures

## 1. Total Sales
TotalSales = SUM(order_details[Sales])

##2. Highest Sales Pizza
Highest Sales Pizza = 
VAR TopPizza =
    TOPN(
        1,
        SUMMARIZE(
            pizza_types,
            pizza_types[name],
            "TotalSales",
            CALCULATE(SUM(order_details[Sales]))
        ),
        [TotalSales],
        DESC
    )
RETURN
    CONCATENATEX(
        TopPizza,
        pizza_types[name],
        ", "
    )

Most Ordered Pizza =
